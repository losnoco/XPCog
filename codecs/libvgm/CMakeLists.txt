# libvgm ships a proper CMake config package -- see ports/libvgm -- so unlike
# most of the overlay-ported libraries here there is no target to assemble.
find_package(libvgm CONFIG REQUIRED GLOBAL)

# The OPL4 sample ROM, compiled in if it is there.
#
# `yrw801.rom` is the 2 MB Yamaha wavetable the YMF278B plays its sample-based
# instruments from. Cog compiles it into its plugin as a 14 MB header of one byte
# per line; kept here as the binary it is, so the repository carries 2 MB rather
# than 14 and the array is generated at build time like Organya's wavetable and
# AdPlug's song database.
#
# The generated translation unit exists whether or not the file does, with a size
# of zero when it does not, so nothing in the decoder is conditional on this and a
# build with the ROM removed still compiles, links and plays -- libvgm warns and
# renders the chip's FM half. See requestFile() in LibVgmDecoder.cpp.
set(YRW801_ROM "${CMAKE_CURRENT_SOURCE_DIR}/data/yrw801.rom")
set(YRW801_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/Yrw801Rom.cpp")

if(EXISTS "${YRW801_ROM}")
    file(READ "${YRW801_ROM}" YRW801_HEX HEX)
    string(LENGTH "${YRW801_HEX}" YRW801_HEX_LENGTH)
    math(EXPR YRW801_LENGTH "${YRW801_HEX_LENGTH} / 2")
    string(REGEX REPLACE "(..)" "0x\\1," YRW801_BYTES "${YRW801_HEX}")
    string(REGEX REPLACE "((0x..,){16})" "\\1\n" YRW801_BYTES "${YRW801_BYTES}")
    message(STATUS "libvgm: OPL4 sample ROM compiled in (${YRW801_LENGTH} bytes)")
else()
    # A one-byte array rather than a zero-length one: a zero-sized array is not
    # standard C++, and the size is what the decoder actually tests.
    set(YRW801_BYTES "0x00,")
    set(YRW801_LENGTH 0)
    message(STATUS "libvgm: no OPL4 sample ROM -- put yrw801.rom in codecs/libvgm/data to add it")
endif()

# `extern` on both definitions, and it is load-bearing rather than tidy: a
# namespace-scope `const` has *internal* linkage in C++, and extern "C" changes
# the mangling without changing that.
file(WRITE "${YRW801_SOURCE}"
"// Generated by codecs/libvgm/CMakeLists.txt. Do not edit.\n"
"extern \"C\" {\n"
"extern const unsigned char kYrw801Rom[];\n"
"extern const unsigned int kYrw801RomSize;\n"
"extern const unsigned char kYrw801Rom[] = {\n${YRW801_BYTES}\n};\n"
"extern const unsigned int kYrw801RomSize = ${YRW801_LENGTH}u;\n"
"}\n")

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${YRW801_ROM}")

xpcog_add_codec(
    NAME     libvgm
    REGISTER xpcog_register_libvgm
    SOURCES  LibVgmDecoder.cpp "${YRW801_SOURCE}"
    DEPS     libvgm::vgm-player libvgm::vgm-emu libvgm::vgm-utils xpcog-codec-common
)
